cmake_minimum_required(VERSION 3.16)
project(laser_projection_ros)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(laser_projection_core REQUIRED)

add_executable(laser_projection_node src/laser_projection_node.cpp)
target_compile_features(laser_projection_node PRIVATE cxx_std_17)
target_compile_options(laser_projection_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(laser_projection_node rclcpp nav_msgs geometry_msgs tf2_ros
  tf2_geometry_msgs visualization_msgs std_msgs laser_projection_core)
install(TARGETS laser_projection_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
